-- Eliminar la base de datos si existe
DROP DATABASE IF EXISTS inventario;

-- Crear la base de datos
CREATE DATABASE inventario;
USE inventario;

-- Tabla sistema operativo
CREATE TABLE sistema_Operativo (
    id_sistemaoperativo INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20)
);

-- Tabla versión de sistema operativo
CREATE TABLE version_SO (
    id_versionso INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20) NOT NULL,
    id_sistemaoperativo INT NOT NULL,
    FOREIGN KEY (id_sistemaoperativo) REFERENCES sistema_Operativo(id_sistemaoperativo)
);

-- Tabla versión de Office
CREATE TABLE version_Office (
    id_versionoffice INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20) NOT NULL
);

-- Tabla RAM
CREATE TABLE ram (
    id_ram INT AUTO_INCREMENT PRIMARY KEY,
    tipo VARCHAR(6) NOT NULL,
    capacidad VARCHAR(6) NOT NULL
);

-- Tabla Disco
CREATE TABLE disco (
    id_disco INT AUTO_INCREMENT PRIMARY KEY,
    capacidad VARCHAR(10) NOT NULL
);

-- Tabla Antivirus
CREATE TABLE antivirus (
    id_antivirus INT AUTO_INCREMENT PRIMARY KEY,
    estado ENUM('Activado', 'Desactivado') NOT NULL
);

-- Tabla Usuario
CREATE TABLE usuario (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL,
    tipo VARCHAR(20) NOT NULL
);

-- Tabla Clasificación
CREATE TABLE clasificacion (
    id_clasificacion INT AUTO_INCREMENT PRIMARY KEY
);

-- Tabla Marca
CREATE TABLE marca (
    id_marca INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20)
);

-- Tabla Modelo
CREATE TABLE modelo (
    id_modelo INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20)
);

-- Tabla Periférico
CREATE TABLE periferico (
    id_periferico INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20)
);

-- Tabla intermedia entre Marca y Periférico
CREATE TABLE marca_periferico (
    id_marca INT NOT NULL,
    id_periferico INT NOT NULL,
    PRIMARY KEY (id_marca, id_periferico),
    FOREIGN KEY (id_marca) REFERENCES marca(id_marca),
    FOREIGN KEY (id_periferico) REFERENCES periferico(id_periferico)
);

-- Tabla intermedia entre Marca y Modelo
CREATE TABLE marca_modelo (
    id_marca INT NOT NULL,
    id_modelo INT NOT NULL,
    PRIMARY KEY (id_marca, id_modelo),
    FOREIGN KEY (id_marca) REFERENCES marca(id_marca),
    FOREIGN KEY (id_modelo) REFERENCES modelo(id_modelo)
);

-- Tabla Serie
CREATE TABLE serie (
    id_serie INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20)
);

-- Tabla intermedia entre Modelo y Serie
CREATE TABLE modelo_serie (
    id_modelo INT NOT NULL,
    id_serie INT NOT NULL,
    PRIMARY KEY (id_modelo, id_serie),
    FOREIGN KEY (id_modelo) REFERENCES modelo(id_modelo),
    FOREIGN KEY (id_serie) REFERENCES serie(id_serie)
);

-- Tabla Equipo
CREATE TABLE equipo (
    id_equipo INT AUTO_INCREMENT PRIMARY KEY,
    id_clasificacion INT NOT NULL,
    inventario VARCHAR(15),
	id_periferico INT NOT NULL,
    
    FOREIGN KEY (id_periferico) REFERENCES periferico(id_periferico),
    FOREIGN KEY (id_clasificacion) REFERENCES clasificacion(id_clasificacion)
);

-- Tabla Computadora
CREATE TABLE computadora (
    id_computadora INT PRIMARY KEY,
    nombre_equipo VARCHAR(20) NOT NULL,
    dominio VARCHAR(20),
    direccion_ip VARCHAR(20),
    id_versionso INT NOT NULL,
    id_versionoffice INT NOT NULL,
    id_ram INT NOT NULL,
    id_disco INT NOT NULL,
    id_antivirus INT NOT NULL,
    FOREIGN KEY (id_computadora) REFERENCES equipo(id_equipo),
    FOREIGN KEY (id_versionso) REFERENCES version_SO(id_versionso),
    FOREIGN KEY (id_versionoffice) REFERENCES version_Office(id_versionoffice),
    FOREIGN KEY (id_ram) REFERENCES ram(id_ram),
    FOREIGN KEY (id_disco) REFERENCES disco(id_disco),
    FOREIGN KEY (id_antivirus) REFERENCES antivirus(id_antivirus)
);

-- Tabla Componente
CREATE TABLE componente (
    id_componente INT PRIMARY KEY,
    id_computadora INT NOT NULL,
    FOREIGN KEY (id_componente) REFERENCES equipo(id_equipo),
    FOREIGN KEY (id_computadora) REFERENCES computadora(id_computadora)
);

-- Tabla Equipo Activo
CREATE TABLE equipo_Activo (
    id_activo INT PRIMARY KEY,
    imagen MEDIUMBLOB NOT NULL,
    edificio VARCHAR(5) NOT NULL,
    aula VARCHAR(5) NOT NULL,
    id_usuario INT NOT NULL,
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),
    FOREIGN KEY (id_activo) REFERENCES clasificacion(id_clasificacion)
);

-- Tabla Equipo Bodega
CREATE TABLE equipo_Bodega (
    id_eqbodega INT PRIMARY KEY,
    FOREIGN KEY (id_eqbodega) REFERENCES clasificacion(id_clasificacion)
);

-- Tabla Equipo Baja
CREATE TABLE equipo_Baja (
    id_eqbaja INT PRIMARY KEY,
    FOREIGN KEY (id_eqbaja) REFERENCES clasificacion(id_clasificacion)
);
